# Telemetria de moto - dependencias Python
#
# Web, tiempo real (SSE), HTTPS y grabacion solo usan la libreria estandar.
# Estos paquetes son para el hardware de la Raspberry Pi; en un portatil no se
# instalan (marcador de plataforma) y todo funciona con --sim.
# Si falta alguno, solo fallan los modulos que lo usan (se ve en la web).
#
# Instalacion en la Pi (Raspberry Pi OS Buster/Bullseye):
#   sudo apt install pigpio python3-pigpio python3-smbus python3-spidev openssl i2c-tools
#   sudo systemctl enable --now pigpiod
#   sudo raspi-config nonint do_i2c 0      # MPU6050 y MLX90614
#   sudo raspi-config nonint do_spi 0      # MAX6675 (hay que reiniciar)
#   dtoverlay=spi1-1cs,cs0_pin=12 en /boot/config.txt   # pantalla por SPI1
#   sudo pip3 install -r requirements.txt  # con sudo: la WS2812 corre como root
#   sudo python3 telemetria.py --detectar  # que hay conectado
#   sudo python3 telemetria.py --https     # web en https://<ip>:8081/
# En Bookworm pip exige un entorno virtual:
#   python3 -m venv --system-site-packages ~/telemetria-venv
#   ~/telemetria-venv/bin/pip install -r requirements.txt  (y usar ese python)
#
# Fuera de pip:
#   pigpiod   demonio de pigpio (apt, arriba); el paquete pip es solo el cliente
#   openssl   genera el certificado de --https (ya viene en Raspberry Pi OS)
#   GPIO18 (WS2812) usa el PWM del audio: dtparam=audio=off en
#   /boot/config.txt (Bookworm: /boot/firmware/config.txt)

# GPIO de todos los modulos (ultrasonidos, ruedas, boton, LED RGB)
pigpio>=1.44 ;sys_platform == "linux" and platform_machine in "armv6l armv7l aarch64"

# I2C: MPU6050 y MLX90614 (python3-smbus de apt tambien vale)
smbus2>=0.4 ;sys_platform == "linux" and platform_machine in "armv6l armv7l aarch64"

# SPI: sondas K con MAX6675
spidev>=3.5 ;sys_platform == "linux" and platform_machine in "armv6l armv7l aarch64"

# Barra de LEDs WS2812 (necesita sudo)
rpi_ws281x>=5.0.0 ; sys_platform == "linux" and platform_machine in "armv6l armv7l aarch64"

# Pantalla OLED (SSD1309 por SPI1); usa RPi.GPIO para los pines DC y RST
luma.oled>=3.8 ; sys_platform == "linux" and platform_machine in "armv6l armv7l aarch64"
RPi.GPIO>=0.7.0 ; sys_platform == "linux" and platform_machine in "armv6l armv7l aarch64"
